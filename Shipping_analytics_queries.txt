USE shipping_analytics;
CREATE TABLE Customers (
    CustomerID VARCHAR(10) PRIMARY KEY,
    CustomerName VARCHAR(100) NOT NULL
);
CREATE TABLE Routes (
    RouteID VARCHAR(10) PRIMARY KEY,
    Origin VARCHAR(100) NOT NULL,
    Destination VARCHAR(100) NOT NULL,
    ShippingMode VARCHAR(20) NOT NULL,
    PlannedTransitDays INT NOT NULL,
    BaseFreightCostUSD DECIMAL(10,2) NOT NULL
);


CREATE TABLE Shipments (
    ShipmentID VARCHAR(15) PRIMARY KEY,
    BookingDate DATE NOT NULL,
    CustomerID VARCHAR(10) NOT NULL,
    RouteID VARCHAR(10) NOT NULL,
    ShippingMode VARCHAR(20) NOT NULL,
    ShipmentType VARCHAR(30),
    CargoWeightKG DECIMAL(10,2) NOT NULL,
    PlannedTransitDays INT NOT NULL,
    PlannedDeliveryDate DATE NOT NULL,
    ActualDeliveryDate DATE,
    ShipmentStatus VARCHAR(20) NOT NULL,
    EstimatedFreightCostUSD DECIMAL(12,2) NOT NULL,
    ActualFreightCostUSD DECIMAL(12,2) NOT NULL,
    RevenueUSD DECIMAL(12,2) NOT NULL,
    DelayDays INT,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID),
    FOREIGN KEY (RouteID) REFERENCES Routes(RouteID)
);

#how many shipments are there for each shipment status?
SELECT
    ShipmentStatus,
    COUNT(*) AS TotalShipments
FROM Shipments
GROUP BY ShipmentStatus
ORDER BY TotalShipments DESC;
#delivery rate
SELECT
    COUNT(*) AS TotalShipments,
    SUM(CASE
        WHEN ShipmentStatus = 'Delivered' THEN 1
        ELSE 0
    END) AS DeliveredShipments,
    ROUND(
        SUM(CASE
            WHEN ShipmentStatus = 'Delivered' THEN 1
            ELSE 0
        END) * 100.0 / COUNT(*),
        2
    ) AS DeliveryRatePercent
FROM Shipments;
#average delivery delays
SELECT
    COUNT(DelayDays) AS ShipmentsWithDelayData,
    ROUND(AVG(DelayDays), 2) AS AverageDelayDays
FROM Shipments;
#number of delayed shipments
SELECT
    COUNT(*) AS DelayedShipments
FROM Shipments
WHERE DelayDays > 0;
#delayed shipment percentage
SELECT
    COUNT(*) AS TotalShipments,
    SUM(CASE
        WHEN DelayDays > 0 THEN 1
        ELSE 0
    END) AS DelayedShipments,
    ROUND(
        SUM(CASE
            WHEN DelayDays > 0 THEN 1
            ELSE 0
        END) * 100.0 / COUNT(*),
        2
    ) AS DelayedShipmentPercent
FROM Shipments;
#frieght cost varience
SELECT
    ShipmentID,
    EstimatedFreightCostUSD,
    ActualFreightCostUSD,
    ROUND(
        ActualFreightCostUSD - EstimatedFreightCostUSD,
        2
    ) AS CostVarianceUSD
FROM Shipments
ORDER BY CostVarianceUSD DESC
LIMIT 10;
#total frieght cost overrun
SELECT
    ROUND(SUM(EstimatedFreightCostUSD), 2) AS TotalEstimatedCost,
    ROUND(SUM(ActualFreightCostUSD), 2) AS TotalActualCost,
    ROUND(
        SUM(ActualFreightCostUSD - EstimatedFreightCostUSD),
        2
    ) AS TotalCostVariance
FROM Shipments;
#frieght cost varience percentage
SELECT
    ROUND(SUM(EstimatedFreightCostUSD), 2)
        AS TotalEstimatedCost,
    ROUND(SUM(ActualFreightCostUSD), 2)
        AS TotalActualCost,
    ROUND(
        (
            SUM(ActualFreightCostUSD)
            - SUM(EstimatedFreightCostUSD)
        ) * 100.0
        / SUM(EstimatedFreightCostUSD),
        2
    ) AS CostVariancePercent
FROM Shipments;
#comparing frieght costs by shipping mode
SELECT
    ShippingMode,
    COUNT(*) AS TotalShipments,
    ROUND(AVG(EstimatedFreightCostUSD), 2)
        AS AvgEstimatedCost,
    ROUND(AVG(ActualFreightCostUSD), 2)
        AS AvgActualCost
FROM Shipments
GROUP BY ShippingMode
ORDER BY AvgActualCost DESC;
#Comparing estimated and actual costs by shipping mode
SELECT
    ShippingMode,
    ROUND(SUM(EstimatedFreightCostUSD), 2)
        AS TotalEstimatedCost,
    ROUND(SUM(ActualFreightCostUSD), 2)
        AS TotalActualCost,
    ROUND(
        (
            SUM(ActualFreightCostUSD)
            - SUM(EstimatedFreightCostUSD)
        ) * 100.0
        / SUM(EstimatedFreightCostUSD),
        2
    ) AS CostVariancePercent
FROM Shipments
GROUP BY ShippingMode
ORDER BY CostVariancePercent DESC;
#shipment delays by shipping mode
SELECT
    ShippingMode,
    COUNT(DelayDays) AS ShipmentsWithDelayData,
    ROUND(AVG(DelayDays), 2) AS AvgDelayDays,
    SUM(CASE
        WHEN DelayDays > 0 THEN 1
        ELSE 0
    END) AS DelayedShipments
FROM Shipments
GROUP BY ShippingMode
ORDER BY AvgDelayDays DESC;
#Calculate the delay rate for each shipping mode
SELECT
    ShippingMode,
    COUNT(*) AS TotalShipments,
    SUM(CASE
        WHEN DelayDays > 0 THEN 1
        ELSE 0
    END) AS DelayedShipments,
    ROUND(
        SUM(CASE
            WHEN DelayDays > 0 THEN 1
            ELSE 0
        END) * 100.0 / COUNT(*),
        2
    ) AS DelayRatePercent
FROM Shipments
GROUP BY ShippingMode
ORDER BY DelayRatePercent DESC;
#freight costs by route
SELECT
    RouteID,
    COUNT(*) AS TotalShipments,
    ROUND(SUM(EstimatedFreightCostUSD), 2)
        AS TotalEstimatedCost,
    ROUND(SUM(ActualFreightCostUSD), 2)
        AS TotalActualCost,
    ROUND(
        SUM(ActualFreightCostUSD - EstimatedFreightCostUSD),
        2
    ) AS CostVarianceUSD
FROM Shipments
GROUP BY RouteID
ORDER BY TotalActualCost DESC;
#route with the highest average cost per shipment
SELECT
    RouteID,
    COUNT(*) AS TotalShipments,
    ROUND(AVG(EstimatedFreightCostUSD), 2)
        AS AvgEstimatedCost,
    ROUND(AVG(ActualFreightCostUSD), 2)
        AS AvgActualCost
FROM Shipments
GROUP BY RouteID
ORDER BY AvgActualCost DESC;
#Combining shipments with customer details
SELECT
    s.ShipmentID,
    c.CustomerName,
    s.ShippingMode,
    s.ShipmentStatus,
    s.ActualFreightCostUSD
FROM Shipments s
INNER JOIN Customers c
    ON s.CustomerID = c.CustomerID
ORDER BY s.ShipmentID
LIMIT 10;
#Shipment, customer, and route analysis
SELECT
    s.ShipmentID,
    c.CustomerName,
    r.Origin,
    r.Destination,
    s.ShippingMode,
    s.ShipmentStatus,
    s.ActualFreightCostUSD
FROM Shipments s
INNER JOIN Customers c
    ON s.CustomerID = c.CustomerID
INNER JOIN Routes r
    ON s.RouteID = r.RouteID
ORDER BY s.ActualFreightCostUSD DESC
LIMIT 10;
